\section{General State-Space Model Framework}

\subsection{Basic State-Space Model Definition}
\textit{Files: \texttt{code/src/core/model\_base.py}, \texttt{report/main.tex}}

State transition (discrete-time):
\begin{equation}
x_k = f(x_{k-1}, w_k), \quad w_k \sim \mathcal{N}(0, Q)
\end{equation}

Observation model:
\begin{equation}
z_k = h(x_k, u_k), \quad u_k \sim \mathcal{N}(0, R)
\end{equation}

\subsection{Bayesian Filtering Equations}
\textit{File: \texttt{report/main.tex}}

Prediction step:
\begin{equation}
p(x_k | z_{1:k-1}) = \int p(x_k | x_{k-1}) p(x_{k-1} | z_{1:k-1}) \, dx_{k-1}
\end{equation}

Update step (Bayes rule):
\begin{equation}
p(x_k | z_{1:k}) = \frac{p(z_k | x_k) p(x_k | z_{1:k-1})}{p(z_k | z_{1:k-1})}
\end{equation}

Marginal likelihood:
\begin{equation}
p(z_k | z_{1:k-1}) = \int p(z_k | x_k) p(x_k | z_{1:k-1}) \, dx_k
\end{equation}

\subsection{Log-Likelihood for Parameter Inference}

Negative log-likelihood (for MLE):
\begin{equation}
L(\theta) = -\sum_{k=1}^{T} \log p_\theta(z_k | z_{1:k-1})
\end{equation}

MAP objective (with prior):
\begin{equation}
L_{\text{MAP}}(\theta) = L(\theta) - \log p(\theta)
\end{equation}

\section{Kalman Filter Family}

\subsection{Standard Kalman Filter (KF)}
\textit{File: \texttt{code/src/filters/kalman/kalman.py}}

Prediction:
\begin{align}
\boldsymbol{m}_{\text{pred}} &= F \boldsymbol{m} \\
P_{\text{pred}} &= F P F^\top + Q
\end{align}

Innovation covariance:
\begin{equation}
S = H P_{\text{pred}} H^\top + R
\end{equation}

Kalman gain:
\begin{equation}
K = P_{\text{pred}} H^\top S^{-1}
\end{equation}

Joseph form covariance update (numerically stable):
\begin{equation}
P_{\text{updated}} = (I - KH) P_{\text{pred}} (I - KH)^\top + K R K^\top
\end{equation}

Log-likelihood per step:
\begin{equation}
\log p(z_k | z_{1:k-1}) = -\frac{1}{2} \left( \log|2\pi S| + \nu^\top S^{-1} \nu \right)
\end{equation}

where $\nu = z_k - H\boldsymbol{m}_{\text{pred}}$ is the innovation.

\subsection{Extended Kalman Filter (EKF)}
\textit{File: \texttt{code/src/filters/kalman/extended\_kalman.py}}

Linearization:
\begin{align}
f(x) &\approx f(\hat{x}) + F (x - \hat{x}), \quad F = \left.\frac{\partial f}{\partial x}\right|_{\hat{x}} \\
h(x) &\approx h(\hat{x}) + H (x - \hat{x}), \quad H = \left.\frac{\partial h}{\partial x}\right|_{\hat{x}}
\end{align}

Prediction:
\begin{align}
\boldsymbol{m}_{\text{pred}} &= f(\boldsymbol{m}) \\
P_{\text{pred}} &= F P F^\top + Q
\end{align}

Update (via Cholesky):
\begin{equation}
L_S K^\top = H P_{\text{pred}} \implies K = (L_S^{-\top} (H P_{\text{pred}})^\top)^\top
\end{equation}

Mean and covariance update:
\begin{align}
\boldsymbol{m}_{\text{updated}} &= \boldsymbol{m}_{\text{pred}} + K (\boldsymbol{z} - h(\boldsymbol{m}_{\text{pred}})) \\
P_{\text{updated}} &= (I - KH) P_{\text{pred}} (I - KH)^\top + K R K^\top
\end{align}

\subsection{Unscented Kalman Filter (UKF)}
\textit{File: \texttt{code/src/filters/kalman/unscented\_kalman.py}}

Sigma point parameters:
\begin{equation}
\lambda = \alpha^2(n + \kappa) - n
\end{equation}

Weights for mean:
\begin{align}
W_m^{(0)} &= \frac{\lambda}{n + \lambda} \\
W_m^{(i)} &= \frac{1}{2(n + \lambda)}, \quad i = 1, \ldots, 2n
\end{align}

Weights for covariance:
\begin{align}
W_c^{(0)} &= W_m^{(0)} + (1 - \alpha^2 + \beta) \\
W_c^{(i)} &= \frac{1}{2(n + \lambda)}, \quad i = 1, \ldots, 2n
\end{align}

Sigma point generation:
\begin{align}
\boldsymbol{\chi}^{(0)} &= \boldsymbol{\mu} \\
\boldsymbol{\chi}^{(i)} &= \boldsymbol{\mu} + \sqrt{(n+\lambda)P}_i, \quad i = 1, \ldots, n \\
\boldsymbol{\chi}^{(n+i)} &= \boldsymbol{\mu} - \sqrt{(n+\lambda)P}_i, \quad i = 1, \ldots, n
\end{align}

Prediction:
\begin{align}
\boldsymbol{\chi}_{\text{pred}}^{(i)} &= f(\boldsymbol{\chi}^{(i)}) \\
\boldsymbol{\mu}_{\text{pred}} &= \sum_i W_m^{(i)} \boldsymbol{\chi}_{\text{pred}}^{(i)} \\
\Sigma_{\text{pred}} &= \sum_i W_c^{(i)} (\boldsymbol{\chi}_{\text{pred}}^{(i)} - \boldsymbol{\mu}_{\text{pred}})(\boldsymbol{\chi}_{\text{pred}}^{(i)} - \boldsymbol{\mu}_{\text{pred}})^\top + Q
\end{align}

\section{Particle Filters}

\subsection{Bootstrap Particle Filter}
\textit{File: \texttt{code/src/filters/particle/bootstrap\_pf\_tf.py}}

Weight update:
\begin{equation}
\log w_t^{(i)} = \log w_{t-1}^{(i)} + \log p(z_t | x_t^{(i)})
\end{equation}

Normalized weights (log-sum-exp):
\begin{equation}
w^{(i)} = \frac{\exp(\log w^{(i)} - \max_j \log w^{(j)})}{\sum_k \exp(\log w^{(k)} - \max_j \log w^{(j)})}
\end{equation}

Effective Sample Size:
\begin{equation}
\text{ESS} = \frac{1}{\sum_i (w_i)^2}
\end{equation}

Weighted mean and covariance:
\begin{align}
\boldsymbol{m}_t &= \sum_i w_t^{(i)} x_t^{(i)} \\
P_t &= \sum_i w_t^{(i)} (x_t^{(i)} - \boldsymbol{m}_t)(x_t^{(i)} - \boldsymbol{m}_t)^\top
\end{align}

\subsection{Conditional SMC}
\textit{File: \texttt{code/src/filters/particle/conditional\_smc.py}}

Ancestor sampling (optional):
\begin{equation}
\log w_{t|T}^{(j)} = \log w_{t-1}^{(j)} + \log \mathcal{N}(x^*_t; f(x_{t-1}^{(j)}), Q)
\end{equation}

Trajectory sampling:
\begin{align}
k_T &\sim \text{Categorical}(w_T) \\
k_t &= \text{ancestors}_{t+1}[k_{t+1}], \quad t = T-1, \ldots, 0
\end{align}

\section{Deterministic Particle Flow Filters}

\subsection{Exact Daum-Huang (EDH) Flow}
\textit{Files: \texttt{code/src/filters/particle/edh\_flow.py}, \texttt{code/src/utils/flow\_params.py}}

Core flow ODE:
\begin{equation}
\frac{d\eta}{d\lambda} = A(\lambda) \eta + b(\lambda), \quad \lambda \in [0, 1]
\end{equation}

Flow parameters (Eq. 10-11 from Li \& Coates 2017):
\begin{equation}
A(\lambda) = -\frac{1}{2} P H^\top (\lambda H P H^\top + R)^{-1} H
\end{equation}

Nonlinear correction:
\begin{equation}
e = h(x) - Hx
\end{equation}

Drift vector:
\begin{equation}
b(\lambda) = (I + 2\lambda A) \left[ (I + \lambda A) P H^\top R^{-1} (z - e) + A \bar{\eta}_0 \right]
\end{equation}

Exponential lambda schedule (Li \& Coates 2017):
\begin{align}
q &= 1.2 \\
\varepsilon_1 &= \frac{1 - q}{1 - q^n} \\
d\lambda_k &= \varepsilon_1 q^k, \quad k = 0, 1, \ldots, n-1
\end{align}

Euler integration:
\begin{equation}
\eta(\lambda + d\lambda) = \eta(\lambda) + d\lambda (A(\lambda) \eta(\lambda) + b(\lambda))
\end{equation}

\subsection{Local EDH (LEDH) Flow}
\textit{File: \texttt{code/src/filters/particle/ledh\_flow.py}}

Per-particle local linearization:
\begin{equation}
H_i = \left.\frac{\partial h}{\partial x}\right|_{x_i}
\end{equation}

Local flow parameters:
\begin{equation}
A_i(\lambda) = -\frac{1}{2} P H_i^\top (\lambda H_i P H_i^\top + R)^{-1} H_i
\end{equation}

With optional regularization:
\begin{equation}
P_{\text{reg}} = P + \text{reg} \cdot \frac{\text{trace}(P)}{n} I
\end{equation}

\subsection{LEDH Invertible Particle Flow Filter (Algorithm 1)}
\textit{File: \texttt{code/src/filters/particle/ledh\_invertible.py}}

Jacobian tracking:
\begin{equation}
M_i = I + d\lambda \cdot A_i
\end{equation}

Log-determinant accumulation:
\begin{equation}
\log\theta_i \leftarrow \log\theta_i + \log|\det(M_i)|
\end{equation}

\section{Stochastic Particle Flow Filters}

\subsection{Stochastic EDH Flow}
\textit{File: \texttt{code/src/filters/particle/stochastic\_edh.py}}

SDE extension (Dai \& Daum 2021, arXiv:2103.09676):
\begin{equation}
dx = [A(\lambda) x + b(\lambda)] d\lambda + \sqrt{Q} \, dW_\lambda
\end{equation}

where $Q = \text{diag}(q_1, \ldots, q_d)$ is diagonal diffusion.

Score-corrected drift:
\begin{equation}
f_{\text{stoch}}(x, \lambda) = f_{\text{det}}(x, \lambda) + \frac{1}{2} Q \nabla_x \log p(x, \lambda)
\end{equation}

Modified flow matrices:
\begin{align}
A &= A_{\text{det}} - \frac{Q}{2} \Sigma(\lambda)^{-1} \\
b &= b_{\text{det}} + \frac{Q}{2} \Sigma(\lambda)^{-1} m(\lambda)
\end{align}

Precision matrix and information vector:
\begin{align}
\Sigma(\lambda)^{-1} &= P_0^{-1} + \lambda H^\top R^{-1} H \\
\Sigma(\lambda)^{-1} m(\lambda) &= P_0^{-1} \bar{\eta}_0 + \lambda H^\top R^{-1} y
\end{align}

Euler-Maruyama:
\begin{equation}
x \leftarrow x + [Ax + b] d\lambda + \sqrt{Q_{\text{diag}} \cdot d\lambda} \, dW
\end{equation}

Fokker-Planck equation:
\begin{equation}
\frac{\partial p}{\partial \lambda} = -\nabla \cdot (f p) + \frac{1}{2} \nabla \cdot (Q Q^\top \nabla p)
\end{equation}

\subsection{SDE with Local Correction}
\textit{File: \texttt{code/src/filters/particle/sde\_local\_correction.py}}

Linearization error:
\begin{equation}
e_0 = h(\bar{\eta}_0) - H \bar{\eta}_0
\end{equation}

Corrected observation:
\begin{equation}
z_{\text{corrected}} = z - e_0
\end{equation}

\section{Stiffness Mitigation via Optimal Schedule}

\subsection{BVP for Optimal Homotopy Schedule}
\textit{File: \texttt{code/src/filters/particle/stochastic\_edh.py}}

Boundary Value Problem:
\begin{align}
\beta'' &= \mu \frac{\partial \kappa}{\partial \beta} \\
\beta(0) &= 0, \quad \beta(1) = 1
\end{align}

Nuclear-norm condition number:
\begin{equation}
\kappa_\nu(M) = \text{tr}(M) \cdot \text{tr}(M^{-1})
\end{equation}

where $M(\beta) = P_0^{-1} + \beta H^\top R^{-1} H$.

Derivative:
\begin{equation}
\frac{\partial \kappa_\nu}{\partial \beta} = \text{tr}(J_{\text{meas}}) \text{tr}(M^{-1}) - \text{tr}(M) \text{tr}(M^{-1} J_{\text{meas}} M^{-1})
\end{equation}

\subsection{Jacobian Stiffness Ratio Schedule}
\textit{File: \texttt{code/src/filters/particle/stochastic\_edh.py}}

Flow Jacobian:
\begin{equation}
F(\beta) = -\frac{1}{2} M(\beta)^{-1} J_{\text{meas}} - \text{diag}(Q/2) M(\beta)
\end{equation}

Stiffness ratio:
\begin{equation}
R_{\text{stiff}} = \frac{\max_i |\Re(\lambda_i(F))|}{\min_i |\Re(\lambda_i(F))|}
\end{equation}

\section{Resampling Methods}

\subsection{Systematic Resampling}
\textit{File: \texttt{code/src/resampling/systematic.py}}

\begin{align}
u &\sim \mathcal{U}(0, 1/N] \\
u_k &= u + k/N, \quad k = 0, 1, \ldots, N-1

\subsection{Soft Resampling}
\textit{File: \texttt{code/src/resampling/soft.py}}

Mixture distribution:
\begin{equation}
q(i) = \alpha w_i + \frac{1-\alpha}{N}
\end{equation}

New weights:
\begin{equation}
w'_i = \frac{w_i}{q(i)}
\end{equation}

\subsection{Entropy-Regularized Optimal Transport}
\textit{File: \texttt{code/src/resampling/ot\_entropy.py}}

Cost matrix:
\begin{equation}
C_{ij} = \frac{\|x_i - y_j\|^2}{2}
\end{equation}

Sinkhorn algorithm:
\begin{align}
\text{softmin}_i(f) &= -\varepsilon \log\left(\sum_j \exp\left(\frac{f_j - C_{ij}}{\varepsilon}\right)\right) \\
\alpha_i^{\text{new}} &= \gamma \alpha_i + (1-\gamma) \text{softmin}_i(\beta + \varepsilon \log w) \\
\beta_j^{\text{new}} &= \gamma \beta_j + (1-\gamma) \text{softmin}_j(\alpha)
\end{align}

Transport matrix:
\begin{equation}
T_{ij} = \exp\left(\frac{\alpha_i + \beta_j - C_{ij}}{\varepsilon}\right) w_i
\end{equation}

\section{Parameter Inference}

\subsection{HMC for Differentiable Filters}
\textit{File: \texttt{code/src/DF/hmc\_runner.py}}

Log-posterior:
\begin{equation}
\log \pi(\theta) = \log p(y_{1:T} | \theta) + \log p(\theta)
\end{equation}

Negative log-posterior (minimized by HMC):
\begin{equation}
U(\theta) = -\log \pi(\theta)
\end{equation}

HMC Hamiltonian:
\begin{equation}
H(p, \theta) = \frac{1}{2} p^\top M^{-1} p + U(\theta)
\end{equation}

Leapfrog integration:
\begin{align}
p &\leftarrow p - \frac{\varepsilon}{2} \nabla_\theta U(\theta) \\
\theta &\leftarrow \theta + \varepsilon M^{-1} p \\
p &\leftarrow p - \frac{\varepsilon}{2} \nabla_\theta U(\theta)
\end{align}

Metropolis acceptance:
\begin{equation}
\alpha = \min(1, \exp(-(H_{\text{end}} - H_{\text{start}})))
\end{equation}

Dual averaging step size adaptation:
\begin{align}
w &= \frac{1}{m + t_0} \\
H_{\text{bar}} &\leftarrow (1-w) H_{\text{bar}} + w(\delta - \text{accept\_prob}) \\
\log \varepsilon &\leftarrow \mu - \frac{\sqrt{m}}{\gamma} H_{\text{bar}}
\end{align}

\subsection{Parameter Bijectors}
\textit{File: \texttt{code/src/DF/parameter\_handler.py}}

Change of variables:
\begin{equation}
\log p(\theta_{\text{unconstrained}}) = \log p(\theta_{\text{constrained}}) + \log|\det(J)|
\end{equation}

Softplus (for $\theta > 0$):
\begin{equation}
\theta_{\text{constrained}} = \text{softplus}(\theta_{\text{unconstrained}}) = \log(1 + \exp(\theta_{\text{unconstrained}}))
\end{equation}

Sigmoid (for $0 < \theta < 1$):
\begin{equation}
\theta_{\text{constrained}} = \frac{1}{1 + \exp(-\theta_{\text{unconstrained}})}
\end{equation}

\subsection{PMMH (Particle Marginal Metropolis-Hastings)}
\textit{File: \texttt{code/src/DF/pmmh\_runner.py}}

Random walk proposal:
\begin{equation}
\theta^* = \theta + \varepsilon, \quad \varepsilon \sim \mathcal{N}(0, \Sigma_{\text{proposal}})
\end{equation}

\section{Summary of Formulas Requiring Hand Derivation}

The following formulas are non-trivial and would typically require careful hand derivation to verify correctness before implementation:

\begin{enumerate}
\item \textbf{EDH flow parameters $A(\lambda)$ and $b(\lambda)$} -- Derived from the log-homotopy equation by assuming a Gaussian ansatz and matching terms. Requires matrix calculus and completing the square in the exponent.

\item \textbf{LEDH local flow with nonlinear correction} -- The drift $b(\lambda)$ includes a correction term $e = h(x) - Hx$ that accounts for linearization error. Must verify the expansion is consistent with the homotopy.

\item \textbf{Stochastic flow score correction} -- Adding $\frac{1}{2}Q\nabla_x \log p$ to the deterministic drift requires deriving the score function of the intermediate Gaussian $p(x,\lambda)$ and verifying it satisfies the Fokker-Planck equation.

\item \textbf{Importance weight formula for invertible flows} -- The full expression $\log w = \log p(z|\eta_1) + \log p(\eta_1|x_{k-1}) + \log\theta - \log p(\eta_0|x_{k-1})$ comes from the change-of-variables formula for the flow map. Requires tracking the Jacobian determinant through the Euler steps.

\item \textbf{Observation Jacobians for each model} -- Each nonlinear model (range-bearing, acoustic, cubic sensor, Kitagawa, stochastic volatility) requires manual differentiation of $h(x)$ to obtain $H = \partial h/\partial x$.

\item \textbf{State Jacobians} -- The Kitagawa model's $\frac{df}{dx} = \frac{1}{2} + \frac{25(1-x^2)}{(1+x^2)^2}$ requires the quotient rule applied to $\frac{25x}{1+x^2}$.

\item \textbf{BVP for optimal homotopy schedule} -- Deriving $\beta'' = \mu \frac{\partial\kappa}{\partial\beta}$ from the stiffness minimization objective, and computing $\frac{\partial\kappa_\nu}{\partial\beta}$ involving traces of matrix products and inverses.

\item \textbf{Stiffness ratio from flow Jacobian eigenvalues} -- Computing $F(\beta) = -\frac{1}{2}M(\beta)^{-1}J_{\text{meas}} - \frac{Q}{2}M(\beta)$ and analyzing its eigenvalue spread.

\item \textbf{Sinkhorn optimal transport} -- The dual formulation of entropy-regularized OT, derivation of the softmin updates, and the transport matrix formula from dual variables.

\item \textbf{HMC leapfrog symplecticity} -- Verifying the leapfrog integrator preserves the symplectic structure and that the Metropolis correction gives the correct stationary distribution.

\item \textbf{Bijector log-determinant Jacobians} -- For the change-of-variables $\log p(\theta_{\text{unc}}) = \log p(\theta_{\text{con}}) + \log|\det J|$, each bijector (softplus, sigmoid) requires computing $\frac{d\theta_{\text{con}}}{d\theta_{\text{unc}}}$.

\item \textbf{Kernel flow SVGD gradient} -- The Stein variational gradient $I_f = \frac{1}{N}\sum_j [K \nabla\log p + \nabla K]$ comes from minimizing KL divergence in the RKHS. Requires functional analysis / Stein's identity.

\item \textbf{Joseph form covariance update} -- Showing $(I-KH)P(I-KH)^\top + KRK^\top$ is equivalent to $(I-KH)P$ but numerically superior, and its limiting behavior as $R \to 0$.

\item \textbf{UKF sigma point weights} -- Deriving the mean and covariance weights from the unscented transform moment-matching conditions.

\item \textbf{PMMH pseudo-marginal correctness} -- Proving that using the particle filter's marginal likelihood estimate $\hat{p}(y|\theta)$ in the MH ratio still targets the correct posterior $p(\theta|y)$.

\item \textbf{Particle Gibbs closed-form conditional} -- Factoring the joint $p(\theta, x | y)$ to obtain $p(\theta | x, y)$ as a product of transition and observation densities conditioned on the sampled trajectory.
\end{enumerate}
